% resume-en-template.tex — English single-page resume template (example persona: Alex Zhang)
% US-letter, classic layout: centered name, small-caps section headers with rules, right-aligned dates. No photo.

% Compile: xelatex resume-en-template.tex (twice)

\documentclass[letterpaper,11pt]{article}

% Section header: small-caps title with full-width rule beneath
\newcommand{\resumesection}[1]{%
  \vspace{4pt}%
  {\large\scshape\textbf{#1}}%
  \vspace{1pt}\hrule height 0.6pt\vspace{4pt}%
}

% Entry heading: bold title left, date right; italic subtitle line
\newcommand{\entryheading}[4]{%
  \textbf{#1}\hfill #2\\[1pt]
  \textit{#3}\hfill \textit{#4}\vspace{2pt}%
}

% Tight bullet list
\newenvironment{entrybullets}{%
  \begin{itemize}[leftmargin=1.4em,itemsep=1pt,topsep=1pt,parsep=0pt]%
}{\end{itemize}\vspace{3pt}}
